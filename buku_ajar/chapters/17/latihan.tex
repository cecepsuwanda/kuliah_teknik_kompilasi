\section*{Latihan dan Refleksi}
\addcontentsline{toc}{section}{Latihan dan Refleksi}

\subsection*{Latihan}
\begin{enumerate}[label=\arabic*.,leftmargin=*]
  \item Petakan lima Sub-CPMK ke bukti belajar Anda.
  \item Analisis kekuatan dan kelemahan proyek Anda.
  \item Rancang satu soal UAS beserta rubriknya.
  \item Jelaskan tiga topik lanjutan yang ingin dipelajari.
  \item Susun rencana belajar satu bulan setelah semester.
  \item Berikan umpan balik konstruktif untuk proyek teman.
  \item Identifikasi satu keputusan desain yang akan Anda ubah.
  \item Tuliskan ringkasan portofolio dalam satu halaman.
\end{enumerate}

\subsection*{Refleksi}
\begin{reflectionbox}
\begin{enumerate}[label=\arabic*.,leftmargin=*]
  \item Kompetensi apa yang paling berkembang?
  \item Kompetensi apa yang masih perlu diperkuat?
  \item Apa bukti konkret dari kemajuan tersebut?
  \item Apa langkah berikutnya?
\end{enumerate}
\end{reflectionbox}
