\section*{Evaluasi}
\addcontentsline{toc}{section}{Evaluasi}

\begin{enumerate}[label=\arabic*.,leftmargin=*]
  \item Petakan lima Sub-CPMK ke bab dan asesmen yang sesuai.
  \item Jelaskan komponen blueprint UTS.
  \item Jelaskan komponen blueprint UAS.
  \item Rancang satu soal studi kasus dan rubriknya.
  \item Jelaskan rubrik komprehensif lima dimensi.
  \item Bagaimana menghindari perhitungan ganda?
  \item Susun kerangka portofolio satu halaman.
  \item Pilih dua topik lanjutan dan jelaskan alasannya.
\end{enumerate}

\subsection*{Kunci dan Rubrik Singkat}
\begin{itemize}[leftmargin=*]
  \item Soal 1--3: ketepatan pemetaan dan blueprint (70\%),
        kelengkapan (30\%).
  \item Soal 4--5: kualitas rancangan dan rubrik (70\%),
        kejelasan (30\%).
  \item Soal 6--8: analisis dan argumentasi (60\%),
        relevansi (40\%).
\end{itemize}
