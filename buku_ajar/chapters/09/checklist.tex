\section*{Checklist Pencapaian Kompetensi}
\addcontentsline{toc}{section}{Checklist Pencapaian Kompetensi}

\begin{checklist}
  \item Saya dapat menjelaskan ambiguitas dengan dua pohon urai.
  \item Saya dapat menjelaskan dangling else dan aturan resolusinya.
  \item Saya dapat menjelaskan precedence dan associativity.
  \item Saya dapat menghapus left recursion langsung.
  \item Saya dapat menghapus left recursion tidak langsung.
  \item Saya dapat melakukan left factoring.
  \item Saya dapat menguji konflik FIRST pada grammar Subset-C.
  \item Saya dapat men-\textit{debug} grammar sampai bebas konflik.
  \item Saya dapat membuktikan kesetaraan bahasa dengan lima string
        uji.
  \item Saya sudah menyelesaikan P2-B.
\end{checklist}
